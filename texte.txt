    function openSmsOtpFlow(contactDisplay) {
      // Génération d'un code aléatoire à 6 chiffres

      document.getElementById('sms-otp-modal').classList.remove('hidden');

    function copyAndFillOtp() {
      if (!state.generatedOtp) return;
      const tempInput = document.createElement('input');
      tempInput.value = state.generatedOtp;
      document.body.appendChild(tempInput);
      tempInput.select();
      document.execCommand('copy');
      document.body.removeChild(tempInput);

      const digits = document.querySelectorAll('.otp-digit');
      state.generatedOtp.split('').forEach((char, idx) => {
        if (digits[idx]) digits[idx].value = char;
      });
      showToast('Code SMS copié et pré-rempli ! Clique sur Continuer.', 'success');
    }

    function startResendCountdown() {
      clearInterval(state.resendInterval);
      let secondsLeft = 30;
      const btn = document.getElementById('resend-sms-btn');
      btn.disabled = true;
      btn.innerHTML = `Renvoyer le SMS (<span id="resend-countdown">${secondsLeft}</span>s)`;

      state.resendInterval = setInterval(() => {
        secondsLeft -= 1;
        const span = document.getElementById('resend-countdown');
        if (span) span.textContent = secondsLeft;
        if (secondsLeft <= 0) {
          clearInterval(state.resendInterval);
          btn.disabled = false;
          btn.textContent = 'Renvoyer le SMS maintenant';
        }
      }, 1000);
    }

    function verifyOtpCode() {
      const digits = document.querySelectorAll('.otp-digit');
      const entered = Array.from(digits).map(d => d.value.trim()).join('');
      const errEl = document.getElementById('otp-error-msg');

      if (entered.length < 6 || entered !== state.generatedOtp) {
        errEl.classList.remove('hidden');
        return;
      }

      errEl.classList.add('hidden');
      closeOtpModal();

      const reg = state.pendingRegistration;
      if (!reg) return;

      if (reg.roleVal === 'artiste') {
        document.getElementById('pub-artist').value = reg.name;
      }

      logSecurityEvent('ENTRÉE', reg.name, `Inscription validée (${reg.role}) via ${reg.contact}.`, 'Vérifié OTP SMS');
      enterPlatform({ name: reg.name, role: reg.role });
      showToast(`Numéro vérifié ! Bienvenue sur Myzing, ${reg.name}.`, 'success');

      if (reg.roleVal === 'artiste') {
        switchTab('studio');
      }

      });

        if (e.key === 'Backspace' && !e.target.value && idx > 0) {
          allInputs[idx - 1].focus();
        }
        if (e.key === 'Enter') {

        }
      });

      input.addEventListener('paste', (e) => {
        e.preventDefault();
        const pasted = (e.clipboardData || window.clipboardData).getData('text').replace(/\D/g, '').slice(0, 6);
        pasted.split('').forEach((ch, i) => {
          if (allInputs[i]) allInputs[i].value = ch;
        });
        if (pasted.length === 6) verifyOtpCode();
      });
    });

    class MyzingAudioEngine {
      constructor() {
        this.ctx = null;
        this.intervalId = null;
        this.step = 0;
        this.isPlaying = false;
      }

      init() {
        if (!this.ctx) {
          const AudioCtx = window.AudioContext || window.webkitAudioContext;
          this.ctx = new AudioCtx();
        }
        if (this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
      }

      playKick(time, punch = 140) {
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(punch, time);
        osc.frequency.exponentialRampToValueAtTime(38, time + 0.14);

        gain.gain.setValueAtTime(0.65, time);
        gain.gain.exponentialRampToValueAtTime(0.001, time + 0.22);

        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(time);
        osc.stop(time + 0.23);
      }

      playRimOrSnare(time, isLogDrum = false) {
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = isLogDrum ? 'triangle' : 'square';
        osc.frequency.setValueAtTime(isLogDrum ? 190 : 320, time);
        osc.frequency.exponentialRampToValueAtTime(isLogDrum ? 65 : 110, time + 0.11);

        gain.gain.setValueAtTime(isLogDrum ? 0.45 : 0.22, time);
        gain.gain.exponentialRampToValueAtTime(0.001, time + 0.13);

        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(time);
        osc.stop(time + 0.14);
      }

      playMelodyNote(time, freq, duration = 0.22) {
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.type = 'sine';
        osc.frequency.setValueAtTime(freq, time);

        gain.gain.setValueAtTime(0.001, time);
        gain.gain.linearRampToValueAtTime(0.18, time + 0.02);
        gain.gain.exponentialRampToValueAtTime(0.001, time + duration);

        osc.connect(gain);
        gain.connect(this.ctx.destination);
        osc.start(time);
        osc.stop(time + duration + 0.02);
      }

      start(vibeIndex = 0) {
        this.stop();
        this.init();
        this.isPlaying = true;
        this.step = 0;

        const patterns = [
          // 0: Groove Afrobeat Solaire
          {
            tempoMs: 138,
            kicks: [0, 6, 8, 14],
            snares: [3, 11],
            notes: { 0: 329.63, 3: 392.00, 6: 440.00, 8: 329.63, 11: 493.88, 14: 392.00 },
            logDrum: false
          },
          // 1: Basse Amapiano Nocturne
          {
            tempoMs: 130,
            kicks: [0, 4, 8, 12],
            snares: [3, 6, 11, 14],
            notes: { 0: 220.00, 4: 261.63, 7: 293.66, 10: 329.63, 14: 261.63 },
            logDrum: true
          },
          // 2: Rap Afro / Drill Percutant
          {
            tempoMs: 118,
            kicks: [0, 7, 10],
            snares: [4, 12, 15],
            notes: { 0: 246.94, 2: 261.63, 6: 329.63, 10: 293.66, 14: 246.94 },
            logDrum: false
          },
          // 3: Mélodie Douce Afro-Soul
          {
            tempoMs: 155,
            kicks: [0, 8],
            snares: [4, 12],
            notes: { 0: 349.23, 2: 440.00, 4: 523.25, 8: 392.00, 10: 493.88, 12: 587.33 },
            logDrum: false
          }
        ];

        const pat = patterns[vibeIndex % patterns.length];

        const tick = () => {
          if (!this.isPlaying) return;
          const now = this.ctx.currentTime;
          const s = this.step % 16;

          if (pat.kicks.includes(s)) this.playKick(now);
          if (pat.snares.includes(s)) this.playRimOrSnare(now, pat.logDrum);
          if (pat.notes[s]) this.playMelodyNote(now, pat.notes[s], 0.26);

          this.step++;
        };

        tick();
        this.intervalId = setInterval(tick, pat.tempoMs);
      }

      stop() {
        this.isPlaying = false;
        if (this.intervalId) {
          clearInterval(this.intervalId);
          this.intervalId = null;
        }
      }
    }

    /* =========================================================================
       CONTRÔLE DE LA LECTURE AUDIO (Fichier MP3/WAV local OU Synthétiseur)
       ========================================================================= */
    function startPlayback(fullTrack = false) {
      const track = state.tracks[state.currentIndex];
      if (!track) return;

      state.isPlaying = true;
      state.fullMode = fullTrack;
      track.plays += 1;

      // Arrêter les deux sources avant de lancer la bonne
      audioEngine.stop();
      htmlAudioPlayer.pause();

      if (track.audioUrl) {
        htmlAudioPlayer.src = track.audioUrl;
        htmlAudioPlayer.currentTime = state.snippetProgress;
        htmlAudioPlayer.play().catch(() => {
          audioEngine.start(track.vibe);
        });
      } else {
        audioEngine.start(track.vibe);
      }

      clearInterval(progressTimer);
      progressTimer = setInterval(() => {
        if (!state.isPlaying) return;
        state.snippetProgress += 0.25;
        const maxDuration = state.fullMode ? 180 : 20;

        if (state.snippetProgress >= maxDuration) {
          state.snippetProgress = 0;
          if (!state.fullMode && state.activeTab === 'shorts') {
            nextShort();
            return;
          }
        }
        updateProgressBarOnly();
      }, 250);

      updateMiniPlayerUI();
      renderActiveShort();
      renderCatalog();
    }

    function stopPlayback() {
      state.isPlaying = false;
      audioEngine.stop();
      htmlAudioPlayer.pause();
      clearInterval(progressTimer);
      updateMiniPlayerUI();
      renderActiveShort();
      renderCatalog();
    }

    function togglePlayCurrent() {
      if (state.isPlaying) {
        stopPlayback();
      } else {
        startPlayback(state.fullMode);
      }
    }

              <div class="w-11 h-11 rounded-full bg-black/50 hover:bg-blue-600/40 text-white backdrop-blur-md border border-white/15 flex items-center justify-center transition">
                <svg class="w-5 h-5" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24">
                  <path stroke-linecap="round" stroke-linejoin="round" d="M8 12h.01M12 12h.01M16 12h.01M21 12c0 4.418-4.03 8-9 8a9.863 9.863 0 01-4.255-.949L3 20l1.395-3.72C3.512 15.042 3 13.574 3 12c0-4.418 4.03-8 9-8s9 3.582 9 8z" />
                </svg>
              </div>
              <span class="text-xs font-bold text-white">${track.comments || 0}</span>
            </button>

            <!-- Écouter en entier -->
            <button onclick="switchToFullSong()" class="flex flex-col items-center gap-1 cursor-pointer" title="Écouter le morceau complet">
              <div class="w-11 h-11 rounded-full ${state.fullMode ? 'bg-sky-500 text-slate-950' : 'bg-black/50 text-sky-300 hover:bg-sky-500/30'} backdrop-blur-md border border-white/15 flex items-center justify-center transition">
                <svg class="w-5 h-5" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24">
                  <path stroke-linecap="round" stroke-linejoin="round" d="M9 19V6l12-3v13M9 19c0 1.105-1.343 2-3 2s-3-.895-3-2 1.343-2 3-2 3 .895 3 2z" />
                </svg>
              </div>
              <span class="text-[10px] font-semibold text-slate-200">Entier</span>
            </button>

            <!-- Partager -->
            <button onclick="shareTrack('${track.id}')" class="flex flex-col items-center gap-1 cursor-pointer">
              <div class="w-11 h-11 rounded-full bg-black/50 hover:bg-blue-600/40 text-white backdrop-blur-md border border-white/15 flex items-center justify-center transition">
                <svg class="w-5 h-5" fill="none" stroke="currentColor" stroke-width="2" viewBox="0 0 24 24">
                  <path stroke-linecap="round" stroke-linejoin="round" d="M8.684 13.342C8.886 12.938 9 12.482 9 12c0-.482-.114-.938-.316-1.342m0 2.684a3 3 0 110-2.684m0 2.684l6.632 3.316m-6.632-6l6.632-3.316m0 0a3 3 0 105.367-2.684 3 3 0 00-5.367 2.684zm0 9.316a3 3 0 105.368 2.684 3 3 0 00-5.368-2.684z" />
                </svg>
              </div>
              <span class="text-[10px] font-semibold text-slate-200">Partager</span>
            </button>

            <!-- Bouton Suivant rapide sur mobile -->
            <button onclick="nextShort()" class="sm:hidden flex flex-col items-center gap-1 cursor-pointer">
              <div class="w-11 h-11 rounded-full bg-blue-600 text-white flex items-center justify-center shadow-lg">
                <svg class="w-5 h-5" fill="none" stroke="currentColor" stroke-width="2.5" viewBox="0 0 24 24">
                  <path stroke-linecap="round" stroke-linejoin="round" d="M19 9l-7 7-7-7" />
                </svg>
              </div>
              <span class="text-[10px] font-bold text-sky-300">Suivant</span>
            </button>
          </div>
        </div>
      `;
    }

    function nextShort() {
      if (state.tracks.length === 0) return;
      state.currentIndex = (state.currentIndex + 1) % state.tracks.length;
      state.snippetProgress = 0;
      if (state.isPlaying) {
        startPlayback(false);
      } else {
        updateMiniPlayerUI();
        renderActiveShort();
      }
    }

    function prevShort() {
      if (state.tracks.length === 0) return;
      state.currentIndex = (state.currentIndex - 1 + state.tracks.length) % state.tracks.length;
      state.snippetProgress = 0;
      if (state.isPlaying) {
        startPlayback(false);
      } else {
        updateMiniPlayerUI();
        renderActiveShort();
      }
    }

    function switchToFullSong() {
      state.fullMode = !state.fullMode;
      if (!state.isPlaying) {
        startPlayback(state.fullMode);
      } else {
        updateMiniPlayerUI();
        renderActiveShort();
      }
      showToast(state.fullMode ? 'Lecture du morceau complet activée.' : 'Retour au mode extrait de 20 secondes.', 'info');
    }

    function toggleLike(id) {
      const track = state.tracks.find(t => t.id === id);
      if (!track) return;
      track.liked = !track.liked;
      track.likes += track.liked ? 1 : -1;
      renderActiveShort();
      renderCatalog();
    }

    function commentTrack(id) {
      const track = state.tracks.find(t => t.id === id);
      if (!track) return;
      track.comments = (track.comments || 0) + 1;
      renderActiveShort();
      showToast(`Ton encouragement a été envoyé à ${track.artist} !`, 'info');
    }

    function toggleFollow(id) {
      const track = state.tracks.find(t => t.id === id);
      if (!track) return;
      track.followed = !track.followed;
      showToast(track.followed ? `Tu suis désormais ${track.artist} !` : `Abonnement retiré pour ${track.artist}.`, 'info');
      renderActiveShort();
    }

    function shareTrack(id) {
      const track = state.tracks.find(t => t.id === id);
      if (!track) return;
      const shareText = `Écoute "${track.title}" de ${track.artist} (${track.country}) sur Myzing !`;
      const tempInput = document.createElement('input');
      tempInput.value = shareText;
      document.body.appendChild(tempInput);
      tempInput.select();
      document.execCommand('copy');
      document.body.removeChild(tempInput);
      showToast('Lien de partage copié dans le presse-papiers !', 'success');
    }

    // Navigation tactile (Swipe Haut/Bas) et molette sur le lecteur Shorts
    let touchStartY = 0;
    const shortContainer = document.getElementById('short-card-container');
    shortContainer.addEventListener('touchstart', (e) => {
      touchStartY = e.changedTouches[0].screenY;
    }, { passive: true });

    shortContainer.addEventListener('touchend', (e) => {
      const touchEndY = e.changedTouches[0].screenY;
      const diff = touchStartY - touchEndY;
      if (Math.abs(diff) > 45) {
        if (diff > 0) nextShort();
        else prevShort();
      }
    }, { passive: true });

    let wheelLock = false;
    shortContainer.addEventListener('wheel', (e) => {
      e.preventDefault();
      if (wheelLock) return;
      wheelLock = true;
      if (e.deltaY > 0) nextShort();
      else prevShort();
      setTimeout(() => { wheelLock = false; }, 450);
    }, { passive: false });

    /* =========================================================================
       ONGLET 3 : STUDIO ARTISTE (Publication MP3/WAV ou Beat généré)
       ========================================================================= */
    document.getElementById('publish-form').addEventListener('submit', (e) => {
      e.preventDefault();
      const title = document.getElementById('pub-title').value.trim();
      const artist = document.getElementById('pub-artist').value.trim();
      const country = document.getElementById('pub-country').value;
      const genre = document.getElementById('pub-genre').value;
      const vibe = parseInt(document.getElementById('pub-vibe').value, 10);
      const gradient = document.querySelector('input[name="pubGradient"]:checked').value;
      const coverInput = document.getElementById('pub-cover-file');
      const audioInput = document.getElementById('pub-audio-file');

      const uploadedAudioUrl = (audioInput.files && audioInput.files[0])
        ? URL.createObjectURL(audioInput.files[0])
        : null;

      const finalizePublish = (coverDataUrl) => {
        const newTrack = {
          id: 'mz-' + Date.now(),
          title,
          artist,
          country,
          genre,
          vibe,
          likes: 1,
          comments: 0,
          liked: true,
          followed: false,
          plays: 1,
          featured: true,
          isNewTalent: true,
          gradient,
          coverUrl: coverDataUrl,
          audioUrl: uploadedAudioUrl
        };

        state.tracks.unshift(newTrack);
        state.currentIndex = 0;
        state.snippetProgress = 0;

        logSecurityEvent(
          'PUBLICATION',
          artist,
          `Nouveau morceau publié : "${title}" (${country} • ${genre}).`,
          uploadedAudioUrl ? 'Fichier audio vérifié' : 'Beat Studio vérifié'
        );

        document.getElementById('publish-form').reset();
        showToast(
          uploadedAudioUrl
            ? `Ton fichier audio "${title}" est en ligne sur Myzing !`
            : `Ton morceau "${title}" est en ligne sur Myzing !`,
          'success'
        );

        switchTab('shorts');
        startPlayback(false);
        renderCatalog();
        renderAdminPanel();
      };

      if (coverInput.files && coverInput.files[0]) {
        const reader = new FileReader();
        reader.onload = (ev) => finalizePublish(ev.target.result);
        reader.readAsDataURL(coverInput.files[0]);
      } else {
        finalizePublish(null);
      }
    });

    function verifyAdminPin(e) {
      e.preventDefault();
      const pin = document.getElementById('admin-pin-input').value.trim();
      if (pin === '1234') {
        state.isAdminUnlocked = true;
        document.getElementById('secret-founder-modal').classList.add('hidden');
        logSecurityEvent('SÉCURITÉ', 'ABESSOLO Deniro', 'Déverrouillage de l’Espace Créateur via PIN secret.', 'Auth Fondateur OK');
        unlockFounderNavigation(true);
        switchTab('admin');
        renderAdminPanel();
        showToast('Bienvenue dans votre espace privé, ABESSOLO Deniro !', 'success');
      } else {
        logSecurityEvent('SÉCURITÉ', 'Inconnu', 'Tentative échouée d’accès PIN Fondateur.', 'Alerte bloquée');
        showToast('Code PIN incorrect.', 'warning');
      }
    }

    function renderAdminPanel() {
      if (!state.isAdminUnlocked) {
        return;
      }

      renderSecurityLogs();

      // Statistiques
      const totalPlays = state.tracks.reduce((acc, t) => acc + t.plays, 0);
      const uniqueArtists = new Set(state.tracks.map(t => t.artist)).size;

      document.getElementById('stat-tracks').textContent = state.tracks.length;
      document.getElementById('stat-plays').textContent = totalPlays.toLocaleString('fr-FR');
      document.getElementById('stat-artists').textContent = uniqueArtists;
      document.getElementById('stat-monetization').textContent = `${state.monetizationCount} / 3`;

      // Pré-remplir le champ de bannière
      document.getElementById('admin-banner-input').value = document.getElementById('banner-text').textContent;

      // Liste de modération des morceaux
      const listEl = document.getElementById('admin-tracks-list');
      listEl.innerHTML = state.tracks.map(track => `
        <div class="flex flex-col sm:flex-row sm:items-center justify-between gap-3 p-3.5 rounded-xl bg-[#080B14] border border-slate-800">
          <div class="min-w-0">
            <div class="flex items-center gap-2">
              <span class="font-bold text-sm text-white truncate">${track.title}</span>
              ${track.featured ? '<span class="px-2 py-0.5 rounded bg-blue-600/25 text-sky-400 text-[10px] font-bold">À la une</span>' : ''}
            </div>
            <div class="text-xs text-slate-400">${track.artist} • ${track.country} • ${track.genre} (${track.plays} écoutes)</div>
          </div>

          <div class="flex items-center gap-2 flex-shrink-0">
            <button onclick="toggleFeaturedTrack('${track.id}')"
              class="px-3 py-1.5 rounded-lg text-xs font-semibold transition cursor-pointer ${
                track.featured
                  ? 'bg-sky-500/20 text-sky-300 border border-sky-400/40'
                  : 'bg-slate-800 text-slate-300 hover:bg-slate-700'
              }">
              ${track.featured ? 'Retirer Une' : 'Mettre à la Une'}
            </button>
            <button onclick="deleteTrackAdmin('${track.id}')"
              class="px-3 py-1.5 rounded-lg bg-red-500/15 hover:bg-red-500/25 text-red-400 text-xs font-semibold transition cursor-pointer">
              Supprimer
            </button>
          </div>
        </div>
      `).join('');
    }

    function updatePlatformBanner() {
      const val = document.getElementById('admin-banner-input').value.trim();
      if (val) {
        document.getElementById('banner-text').textContent = val;
        showToast('Bannière d’annonce mise à jour en direct !', 'success');
      }
    }

    function toggleMonetization(checkbox, label) {
      const toggles = document.querySelectorAll('.monetization-toggle:checked');
      state.monetizationCount = toggles.length;
      document.getElementById('stat-monetization').textContent = `${state.monetizationCount} / 3`;
      showToast(
        checkbox.checked
          ? `Module "${label}" activé dans la configuration.`
          : `Module "${label}" mis en veille.`,
        'info'
      );
    }

    function toggleFeaturedTrack(id) {
      const track = state.tracks.find(t => t.id === id);
      if (!track) return;
      track.featured = !track.featured;
      renderAdminPanel();
      renderActiveShort();
      showToast(track.featured ? `"${track.title}" mis à la une !` : `"${track.title}" retiré de la une.`, 'info');
    }

    function deleteTrackAdmin(id) {
      if (state.tracks.length <= 1) {
        showToast('Garde au moins un morceau sur la plateforme.', 'warning');
        return;
      }
      state.tracks = state.tracks.filter(t => t.id !== id);
      if (state.currentIndex >= state.tracks.length) {
        state.currentIndex = 0;
      }
      renderAdminPanel();
      renderActiveShort();
      renderCatalog();
      updateMiniPlayerUI();
      showToast('Morceau supprimé de la plateforme.', 'info');
    }
